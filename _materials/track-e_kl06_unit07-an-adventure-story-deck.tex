\begin{frame}{Learning objectives}
\begin{itemize}
  \item \textit{I can read a short adventure story and identify setting, characters, plot.}
  \item \textit{I can use past simple and past continuous together with while} and \textit{when}.
  \item \textit{I can write a 100-word continuation in past tense.}
\end{itemize}
\end{frame}

\begin{frame}{Lead-in}
\large Sam found an old envelope behind the bookshelf. The paper was thick and yellow. Inside there was one page, half a map, and the signature \textit{Captain Cody — 1782}. The map showed an island that did not appear on any modern atlas. Sam decided not to mention it to anyone. Yet.
\end{frame}

\begin{frame}{Key points}
\small
\begin{itemize}
  \item \textbf{While} + past continuous (longer action):
  \item \textbf{When} + past simple (event):
  \item Form: \textit{was/were + -ing}.
\end{itemize}
\end{frame}

\begin{frame}{Practise}
\small
\begin{itemize}
  \item Past continuous: I \_\_\_\_\_\_\_\_\_\textit{ (read). Sam }\_\_\_\_\_\_\_\_\textit{ (walk) home. They }\_\_\_\_\_\_\_\_\_ (look) at the map.
  \item Choose \textit{while} or \textit{when}: \_\_\textit{ I was reading, the phone rang. }\_\_ Sam was walking, it started to rain.
  \item Build 4 sentences combining past continuous + past simple with \textit{when} and \textit{while}.
\end{itemize}
\end{frame}

\begin{frame}{Your turn}
\large \textbf{Story continuation.} Write 100 words continuing \textit{Captain Cody's Letter}. What happens next? Stay in past simple. Use one \textit{while} and one \textit{when}.
\end{frame}

\begin{frame}{Reflect}
\small
\begin{itemize}
  \item I can identify setting, characters, plot in a short story.
  \item I can use past simple and past continuous together.
  \item I can write a 100-word continuation.
\end{itemize}
\end{frame}
